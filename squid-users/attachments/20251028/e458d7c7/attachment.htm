<p class="MsoNormal"><o:p> </o:p></p>
<p class="MsoNormal"># add diladele apt key<o:p></o:p></p>
<p class="MsoNormal">wget -qO - https://packages.diladele.com/diladele_pub.asc | sudo apt-key add -<o:p></o:p></p>
<p class="MsoNormal"><o:p> </o:p></p>
<p class="MsoNormal"># add new repo<o:p></o:p></p>
<p class="MsoNormal">echo "deb https://squid72.diladele.com/ubuntu/ noble main" \<o:p></o:p></p>
<p class="MsoNormal">    > /etc/apt/sources.list.d/squid72.diladele.com.list<o:p></o:p></p>
<p class="MsoNormal"><o:p> </o:p></p>
<p class="MsoNormal"># and install<o:p></o:p></p>
<p class="MsoNormal">apt update && apt install -y \<o:p></o:p></p>
<p class="MsoNormal">    squid-common \<o:p></o:p></p>
<p class="MsoNormal">    squid-openssl \<o:p></o:p></p>
